package cloudscheduler;

/**
 * This class represents a single compute request submitted by a cloud client.
 * Each request stores who submitted it, what resource they need, and what
 * priority level they should get in the scheduler queue.
 * 
 * I split the priority into two values: basePriority (what the user set at
 * submission time) and currentPriority (which can increase over time due to aging).
 * This way the original priority is never lost.
 * 
 * The static compare() method at the bottom is used by both MaxHeap and the
 * display sort so they both use the exact same ordering rules.
 * 
 * Ordering from highest to lowest:
 *   1. Higher currentPriority wins
 *   2. Earlier arrival time wins if priority is equal
 *   3. Lower submission number wins if arrival time is also equal (strict FIFO)
 */
public class ComputeRequest {

    // -- Fields that are set once at submission and never change --

    /** Who submitted this request. */
    final String clientId;

    /** Either "CPU" or "GPU". */
    final String resourceType;

    /** The priority the user gave when submitting. This never changes. */
    final int basePriority;

    /** The simulated clock time when this request was added to the queue. */
    final long entryTimestamp;

    /**
     * A unique counter value that goes up by 1 with each submission.
     * Used as a last tiebreaker so the queue is strictly fair (FIFO).
     */
    final long submissionSequenceNumber;

    // -- Fields that can change as the scheduler runs --

    /**
     * The working priority used for scheduling. Starts equal to basePriority
     * but can grow through aging. It never goes above 10 and never decreases.
     */
    int currentPriority;

    /** Becomes true the first time this request gets an aging boost. Used for display. */
    boolean hasBeenAged;

    // -- Constructor --

    ComputeRequest(String clientId,
                   String resourceType,
                   int basePriority,
                   long entryTimestamp,
                   long submissionSequenceNumber) {

        this.clientId                = clientId;
        this.resourceType            = resourceType;
        this.basePriority            = basePriority;
        this.currentPriority         = basePriority;
        this.entryTimestamp          = entryTimestamp;
        this.submissionSequenceNumber = submissionSequenceNumber;
        this.hasBeenAged             = false;
    }

    /** Simple check to see if this request has ever received an aging boost. */
    boolean isAged() {
        return hasBeenAged;
    }

    /**
     * Compares two requests to decide which one should be served first.
     * Returns a positive number if 'a' should come before 'b'.
     * 
     * Rule 1: higher currentPriority wins.
     * Rule 2: if equal priority, whoever arrived first wins.
     * Rule 3: if arrival time is also equal, the one submitted first wins.
     */
    static int compare(ComputeRequest a, ComputeRequest b) {

        // Rule 1: higher priority goes first
        if (a.currentPriority != b.currentPriority) {
            return a.currentPriority - b.currentPriority;
        }

        // Rule 2: earlier arrival time goes first (positive means a arrived earlier)
        if (a.entryTimestamp != b.entryTimestamp) {
            return (a.entryTimestamp < b.entryTimestamp) ? 1 : -1;
        }

        // Rule 3: lower sequence number goes first (the one submitted first)
        if (a.submissionSequenceNumber != b.submissionSequenceNumber) {
            return (a.submissionSequenceNumber < b.submissionSequenceNumber) ? 1 : -1;
        }

        return 0;
    }

    /**
     * Builds a single readable line summarising this request.
     * Used by both the allocation output and the queue display.
     */
    String buildSummaryLine(long currentSimulatedTime) {
        long waitingSeconds = currentSimulatedTime - entryTimestamp;
        String agedMarker   = hasBeenAged ? " *** AGED ***" : "";
        return String.format(
            "Client: %-10s | Type: %-3s | Base: %2d | Priority: %2d | Wait: %3ds%s",
            clientId, resourceType, basePriority, currentPriority,
            waitingSeconds, agedMarker
        );
    }
}

package cloudscheduler;

/**
 * This is a custom Binary Max-Heap that I built from scratch to store
 * ComputeRequest objects. I built it myself instead of using Java's
 * PriorityQueue because the assignment requires no built-in data structures.
 * 
 * The heap stores everything in a plain array. The highest-priority request
 * always sits at index 0 (the root), so extracting it is fast.
 * 
 * How the array positions work:
 *   - Parent of index i  is at  (i - 1) / 2
 *   - Left child of i   is at  2*i + 1
 *   - Right child of i  is at  2*i + 2
 * 
 * The heap rule: every parent must have higher (or equal) priority than
 * both of its children, using ComputeRequest.compare() to decide.
 * 
 * Time costs:
 *   - offer() (insert)  : O(log n)
 *   - poll()  (remove)  : O(log n)
 *   - peek()  (look)    : O(1)
 */
public class MaxHeap {

    private ComputeRequest[] data;
    private int size;
    private static final int INITIAL_CAPACITY = 16;

    MaxHeap() {
        data = new ComputeRequest[INITIAL_CAPACITY];
        size = 0;
    }

    /** How many requests are currently in the heap. */
    int size() {
        return size;
    }

    /** Returns true if the heap has no requests in it. */
    boolean isEmpty() {
        return size == 0;
    }

    /**
     * Adds a new request to the heap.
     * The request goes to the end of the array and then bubbles up
     * (siftUp) until it finds its correct position.
     * Time: O(log n)
     */
    void offer(ComputeRequest request) {
        if (size == data.length) {
            resize();
        }
        data[size] = request;
        siftUp(size);
        size++;
    }

    /**
     * Removes and returns the highest-priority request (the one at the root).
     * The last element in the array moves to the root, then sinks down
     * (siftDown) to find its correct position.
     * Returns null if the heap is empty.
     * Time: O(log n)
     */
    ComputeRequest poll() {
        if (size == 0) {
            return null;
        }
        ComputeRequest top = data[0];

        // Move the last element to the root to fill the gap.
        size--;
        data[0] = data[size];
        data[size] = null; // let GC reclaim memory

        if (size > 0) {
            siftDown(0);
        }
        return top;
    }

    /**
     * Returns the highest-priority request without removing it from the heap.
     * Returns null if the heap is empty.
     * Time: O(1)
     */
    ComputeRequest peek() {
        return (size == 0) ? null : data[0];
    }

    /**
     * Returns a copy of all the elements currently in the heap as a plain array.
     * The order matches the internal heap array, not scheduling order.
     * The heap itself is not modified.
     * Time: O(n)
     */
    ComputeRequest[] toArray() {
        ComputeRequest[] copy = new ComputeRequest[size];
        for (int i = 0; i < size; i++) {
            copy[i] = data[i];
        }
        return copy;
    }

    /**
     * Empties the heap completely.
     * Time: O(n)
     */
    void clear() {
        for (int i = 0; i < size; i++) {
            data[i] = null; // let GC reclaim memory
        }
        size = 0;
    }

    // -- Private heap mechanics --

    /**
     * Bubbles the element at 'index' upward until the heap rule is satisfied.
     * Used after adding a new element at the end.
     */
    private void siftUp(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;
            // If this element has higher priority than its parent, swap them.
            if (ComputeRequest.compare(data[index], data[parent]) > 0) {
                swap(index, parent);
                index = parent;
            } else {
                break; // heap rule is restored, stop
            }
        }
    }

    /**
     * Sinks the element at 'index' downward until the heap rule is satisfied.
     * Used after removing the root and placing the last element there.
     */
    private void siftDown(int index) {
        while (true) {
            int left    = 2 * index + 1;
            int right   = 2 * index + 2;
            int largest = index; // track which of parent/left/right has highest priority

            // Check if the left child has higher priority than the current largest.
            if (left < size && ComputeRequest.compare(data[left], data[largest]) > 0) {
                largest = left;
            }

            // Check if the right child beats even that.
            if (right < size && ComputeRequest.compare(data[right], data[largest]) > 0) {
                largest = right;
            }

            if (largest == index) {
                break; // the element is already in the right place
            }

            swap(index, largest);
            index = largest;
        }
    }

    /** Swaps two elements in the backing array. */
    private void swap(int i, int j) {
        ComputeRequest temp = data[i];
        data[i] = data[j];
        data[j] = temp;
    }

    /**
     * When the backing array is full, this doubles its size and copies
     * all existing elements over. This keeps insertion amortised O(log n).
     */
    private void resize() {
        ComputeRequest[] larger = new ComputeRequest[data.length * 2];
        for (int i = 0; i < data.length; i++) {
            larger[i] = data[i];
        }
        data = larger;
    }
}

package cloudscheduler;

/**
 * This class is the brain of the scheduler. It owns the MaxHeap and
 * implements the four main operations (A, B, C, D) that the assignment requires.
 * 
 * I separated the logic here from CloudSchedulerApp (the menu/UI) so that
 * the actual scheduling algorithm is independent and easy to test on its own.
 * 
 * Data structure used: Custom Binary Max-Heap (see MaxHeap.java)
 *   - O(log n) to insert a new request
 *   - O(log n) to extract the highest-priority request
 * 
 * The aging formula I used to prevent starvation:
 *   agingBonus  = (waitingTime - 1) / agingIntervalSeconds   (integer division)
 *   newPriority = min(10, basePriority + agingBonus)
 * 
 * The "(waitingTime - 1)" makes the condition "strictly more than X seconds":
 *   If X=10 and wait=10 → (10-1)/10 = 0 → no boost yet  ✓
 *   If X=10 and wait=11 → (11-1)/10 = 1 → one boost      ✓
 *   If X=10 and wait=21 → (21-1)/10 = 2 → two boosts     ✓
 * 
 * Time complexities:
 *   submitComputeRequest            : O(log n)
 *   advanceClockAndApplyAgingBoosts : O(n log n)
 *   allocateTopNHighPriorityRequests: O(k log n)  where k = requests taken
 *   displayScheduledQueue           : O(n log n)  due to insertion sort snapshot
 */
public class PrioritySchedulerEngine {

    static final int MAX_PRIORITY = 10;
    static final int MIN_PRIORITY = 1;

    /** The heap that holds all pending requests. */
    private final MaxHeap schedulingHeap;

    /** The simulated clock in seconds. Starts at 0. */
    private long simulatedClockSeconds;

    /** Increments by 1 with every submission to guarantee strict FIFO ordering. */
    private long submissionSequenceCounter;

    /** How many seconds a request must wait before getting a priority boost. */
    private final int agingIntervalSeconds;

    PrioritySchedulerEngine(int agingIntervalSeconds) {
        this.schedulingHeap           = new MaxHeap();
        this.agingIntervalSeconds     = agingIntervalSeconds;
        this.simulatedClockSeconds    = 0L;
        this.submissionSequenceCounter = 0L;
    }

    // -- Accessor methods used by the UI to show status in the menu --
    long getSimulatedClockSeconds() { return simulatedClockSeconds; }
    int  getPendingRequestCount()   { return schedulingHeap.size(); }
    int  getAgingIntervalSeconds()  { return agingIntervalSeconds; }

    // -- Operation A: Submit a new compute request --

    /**
     * Creates a ComputeRequest object with the current clock time and a
     * unique sequence number, then inserts it into the max-heap.
     * Time: O(log n) because of the siftUp inside offer().
     */
    void submitComputeRequest(String clientId,
                              String resourceType,
                              int basePriority) {

        ComputeRequest newRequest = new ComputeRequest(
            clientId,
            resourceType,
            basePriority,
            simulatedClockSeconds,
            submissionSequenceCounter++
        );

        schedulingHeap.offer(newRequest);

        printSectionDivider();
        System.out.println("  [SUBMITTED SUCCESSFULLY]");
        System.out.println("  " + newRequest.buildSummaryLine(simulatedClockSeconds));
        printSectionDivider();
    }

    // -- Operation B: Advance the clock and apply aging boosts --

    /**
     * Moves the simulated clock forward and recalculates the priority for
     * every waiting request using the aging formula.
     * 
     * Why I drain and rebuild the heap:
     * The heap does not automatically reposition elements when their priority
     * changes. So I pull everything out into a temporary array, update each
     * request's priority, then put them all back in with offer() so the
     * heap reorganises itself correctly.
     * 
     * Time: O(n log n) because each of the n requests goes through offer() which is O(log n).
     */
    void advanceClockAndApplyAgingBoosts(int secondsToAdvance) {

        simulatedClockSeconds += secondsToAdvance;

        if (schedulingHeap.isEmpty()) {
            printSectionDivider();
            System.out.printf(
                "  [CLOCK] Advanced to %ds — queue is empty; no aging applied.%n",
                simulatedClockSeconds);
            printSectionDivider();
            return;
        }

        // Step 1: pull all requests out into a flat array.
        ComputeRequest[] buffer = schedulingHeap.toArray();
        schedulingHeap.clear();

        int boostCount = 0;

        // Step 2: recalculate each request's priority, then re-add to the heap.
        for (int i = 0; i < buffer.length; i++) {
            ComputeRequest request = buffer[i];
            long waitingTime = simulatedClockSeconds - request.entryTimestamp;

            // Aging formula: the -1 enforces "strictly more than X seconds"
            int agingBonus  = (int) ((waitingTime - 1) / agingIntervalSeconds);
            int newPriority = Math.min(MAX_PRIORITY, request.basePriority + agingBonus);

            if (newPriority > request.currentPriority) {
                request.hasBeenAged = true;
                boostCount++;
            }
            request.currentPriority = newPriority;

            schedulingHeap.offer(request); // O(log n) sift-up
        }

        printSectionDivider();
        System.out.printf(
            "  [CLOCK]  Simulated time is now %ds  |  Aging interval: every %ds%n",
            simulatedClockSeconds, agingIntervalSeconds);
        System.out.printf(
            "  [AGING]  %d of %d request(s) received a priority boost.%n",
            boostCount, buffer.length);
        printSectionDivider();
    }

    // -- Operation C: Remove and serve the top N requests --

    /**
     * Repeatedly extracts the highest-priority request from the heap
     * and prints it. Stops early if the heap runs empty before reaching N.
     * Time: O(k log n) where k is how many requests we actually extract.
     */
    void allocateTopNHighPriorityRequests(int numberOfRequestsToAllocate) {

        if (schedulingHeap.isEmpty()) {
            printSectionDivider();
            System.out.println("  [ALLOCATE] Queue is empty — no requests to allocate.");
            printSectionDivider();
            return;
        }

        int actualCount = Math.min(numberOfRequestsToAllocate, schedulingHeap.size());
        printSectionDivider();
        System.out.printf("  [ALLOCATING TOP %d REQUEST(S)]%n%n", actualCount);

        for (int rank = 1; rank <= actualCount; rank++) {
            ComputeRequest served = schedulingHeap.poll(); // O(log n) sift-down
            System.out.printf("  [%d] %s%n", rank,
                              served.buildSummaryLine(simulatedClockSeconds));
        }

        System.out.printf("%n  Allocation complete. %d request(s) still pending.%n",
                          schedulingHeap.size());
        printSectionDivider();
    }

    // -- Operation D: Display the full queue without removing anything --

    /**
     * Shows every request in scheduling order without touching the live heap.
     * 
     * How it works:
     *   1. Copy the heap contents into a plain array (doesn't touch the heap).
     *   2. Sort that array using insertion sort so they appear in priority order.
     *   3. Print the sorted list. The real heap is untouched throughout.
     * 
     * I chose insertion sort because it needs no extra data structures and
     * the queue sizes in this simulation are small enough that O(n²) is fine.
     */
    void displayScheduledQueue() {

        printSectionDivider();
        System.out.printf(
            "  [QUEUE STATUS]  Clock=%ds | Aging every %ds | %d request(s) pending%n",
            simulatedClockSeconds, agingIntervalSeconds, schedulingHeap.size());

        if (schedulingHeap.isEmpty()) {
            System.out.println("  Queue is currently empty.");
            printSectionDivider();
            return;
        }

        // Take a snapshot — this does NOT change the heap.
        ComputeRequest[] snapshot = schedulingHeap.toArray();

        // Sort the snapshot so index 0 = the request that would be served first.
        insertionSortBySchedulingOrder(snapshot);

        System.out.println();
        System.out.printf("  %-4s %-12s %-5s %6s %9s %9s %-14s%n",
            "#", "Client ID", "Type", "Base", "Priority", "Waiting", "Status");
        System.out.println("  " + repeat('-', 68));

        int agedCount = 0;
        for (int i = 0; i < snapshot.length; i++) {
            ComputeRequest r    = snapshot[i];
            long waitSeconds    = simulatedClockSeconds - r.entryTimestamp;
            String statusLabel  = r.isAged() ? "*** AGED ***" : "normal";
            String nextPointer  = (i == 0) ? " <- NEXT" : "";

            System.out.printf("  %-4d %-12s %-5s %6d %9d %8ds %-14s%s%n",
                i + 1, r.clientId, r.resourceType, r.basePriority,
                r.currentPriority, waitSeconds, statusLabel, nextPointer);

            if (r.isAged()) agedCount++;
        }

        System.out.println();
        System.out.printf("  Summary: %d aged | %d normal | %d total%n",
            agedCount, snapshot.length - agedCount, snapshot.length);
        printSectionDivider();
    }

    // -- Private sort helper --

    /**
     * Sorts the array so that the highest-priority request is at index 0.
     * Works just like sorting numbers in descending order, but uses
     * ComputeRequest.compare() to decide the order.
     */
    private static void insertionSortBySchedulingOrder(ComputeRequest[] arr) {
        for (int i = 1; i < arr.length; i++) {
            ComputeRequest key = arr[i];
            int j = i - 1;
            // Shift elements with lower priority one position to the right.
            while (j >= 0 && ComputeRequest.compare(arr[j], key) < 0) {
                arr[j + 1] = arr[j];
                j--;
            }
            arr[j + 1] = key;
        }
    }

    // -- Private string utility (Java 8 does not have String.repeat) --

    private static String repeat(char ch, int times) {
        char[] chars = new char[times];
        for (int i = 0; i < times; i++) chars[i] = ch;
        return new String(chars);
    }

    /** Shared divider line used by all four operations for consistent output formatting. */
    static void printSectionDivider() {
        System.out.println("  " + "─────────────────────────────────────────────────────────────────────");
    }
}
part-A/CloudSchedulerApp.java 
package cloudscheduler;

import java.util.Scanner;

/**
 * This is the main entry point of the Cloud Scheduler application.
 * It handles all the user interaction - showing the menu, reading input,
 * and calling the appropriate scheduling methods.
 * 
 * I kept all the actual scheduling logic inside PrioritySchedulerEngine
 * so this class stays clean and only deals with the console UI.
 * 
 * The only import I used in the whole project is java.util.Scanner right here,
 * everything else like String and Math are built into Java automatically.
 */
public class CloudSchedulerApp {

    public static void main(String[] args) {

        Scanner userInputScanner = new Scanner(System.in);

        printWelcomeBanner();

        // Ask the user to set the aging interval before starting.
        // If they press Enter without typing anything, it defaults to 10 seconds.
        System.out.print("  Set aging interval X in seconds (default = 10): ");
        String rawAgingInput = userInputScanner.nextLine().trim();

        int chosenAgingInterval = 10;
        try {
            int parsed = Integer.parseInt(rawAgingInput);
            if (parsed > 0) chosenAgingInterval = parsed;
        } catch (NumberFormatException ignored) {}

        System.out.printf("%n  Scheduler started. Aging interval X = %ds.%n",
                          chosenAgingInterval);
        System.out.println("  Priority boosts after STRICTLY MORE THAN X seconds waiting.");
        System.out.println("  Priority is capped at " + PrioritySchedulerEngine.MAX_PRIORITY
                           + " and never decreases.");

        PrioritySchedulerEngine schedulerEngine =
            new PrioritySchedulerEngine(chosenAgingInterval);

        // Keep showing the menu until the user chooses to exit.
        while (true) {

            printMenuOptions(schedulerEngine);
            String userChoice = userInputScanner.nextLine().trim().toUpperCase();

            switch (userChoice) {

                case "A": {
                    System.out.println();
                    System.out.println("  --- SUBMIT NEW REQUEST -------------------------------------------");
                    String clientId     = readNonEmptyString(userInputScanner, "Client ID: ");
                    String resourceType = readResourceType(userInputScanner);
                    int basePriority    = readIntegerInRange(
                                            userInputScanner,
                                            "Base Priority (1-10): ",
                                            PrioritySchedulerEngine.MIN_PRIORITY,
                                            PrioritySchedulerEngine.MAX_PRIORITY);
                    schedulerEngine.submitComputeRequest(clientId, resourceType, basePriority);
                    break;
                }

                case "B": {
                    System.out.println();
                    System.out.println("  --- SIMULATE TIME ------------------------------------------------");
                    int secondsToAdvance = readIntegerInRange(
                                            userInputScanner,
                                            "Seconds to advance (> 0): ",
                                            1, Integer.MAX_VALUE);
                    schedulerEngine.advanceClockAndApplyAgingBoosts(secondsToAdvance);
                    break;
                }

                case "C": {
                    System.out.println();
                    System.out.println("  --- ALLOCATE TOP N REQUESTS --------------------------------------");
                    int n = readIntegerInRange(
                                userInputScanner,
                                "Number of requests to allocate (N >= 1): ",
                                1, Integer.MAX_VALUE);
                    schedulerEngine.allocateTopNHighPriorityRequests(n);
                    break;
                }

                case "D": {
                    schedulerEngine.displayScheduledQueue();
                    break;
                }

                case "E":
                case "EXIT": {
                    PrioritySchedulerEngine.printSectionDivider();
                    System.out.println("  Scheduler stopped. All pending requests discarded.");
                    PrioritySchedulerEngine.printSectionDivider();
                    userInputScanner.close();
                    return;
                }

                default:
                    System.out.println(
                        "  [ERROR] Invalid option. Please choose A, B, C, D, or E.");
            }
        }
    }

    // -- UI helper methods below -- these only deal with printing and reading input --

    private static void printWelcomeBanner() {
        System.out.println();
        System.out.println("  +----------------------------------------------+");
        System.out.println("  |     CLOUD COMPUTE INSTANCE SCHEDULER         |");
        System.out.println("  |   Priority Scheduling + Starvation Aging     |");
        System.out.println("  |   Data Structures: Custom Binary Max-Heap    |");
        System.out.println("  |   Import used: java.util.Scanner ONLY        |");
        System.out.println("  +----------------------------------------------+");
        System.out.println();
    }

    private static void printMenuOptions(PrioritySchedulerEngine engine) {
        System.out.println();
        System.out.println("  +---------------------------------------------+");
        System.out.printf ("  |  Simulated Clock  : %-22ds |%n",
                            engine.getSimulatedClockSeconds());
        System.out.printf ("  |  Aging Interval   : every %-16ds |%n",
                            engine.getAgingIntervalSeconds());
        System.out.printf ("  |  Queue Size       : %-22d |%n",
                            engine.getPendingRequestCount());
        System.out.println("  +---------------------------------------------+");
        System.out.println("  |  A -- Submit New Request                    |");
        System.out.println("  |  B -- Simulate Time (Apply Aging)           |");
        System.out.println("  |  C -- Allocate Top N Requests               |");
        System.out.println("  |  D -- Display Current Queue                 |");
        System.out.println("  |  E -- Exit                                  |");
        System.out.println("  +---------------------------------------------+");
        System.out.print("  Choice: ");
    }

    /** Keeps asking until the user types something that is not blank. */
    private static String readNonEmptyString(Scanner scanner, String prompt) {
        while (true) {
            System.out.print("  " + prompt);
            String value = scanner.nextLine().trim();
            if (!value.isEmpty()) return value;
            System.out.println("  [ERROR] Input cannot be empty. Please try again.");
        }
    }

    /** Keeps asking until the user types CPU or GPU (case doesn't matter). */
    private static String readResourceType(Scanner scanner) {
        while (true) {
            System.out.print("  Resource Type (CPU / GPU): ");
            String value = scanner.nextLine().trim().toUpperCase();
            if (value.equals("CPU") || value.equals("GPU")) return value;
            System.out.println("  [ERROR] Resource type must be CPU or GPU.");
        }
    }

    /** Keeps asking until the user enters a whole number between min and max. */
    private static int readIntegerInRange(Scanner scanner, String prompt, int min, int max) {
        while (true) {
            System.out.print("  " + prompt);
            try {
                int value = Integer.parseInt(scanner.nextLine().trim());
                if (value >= min && value <= max) return value;
            } catch (NumberFormatException ignored) {}
            System.out.printf(
                "  [ERROR] Please enter an integer between %d and %d.%n", min, max);
        }
    }
}

package huffman;

/**
 * This class represents a single node inside the Huffman binary tree.
 * 
 * There are two kinds of nodes:
 *   - Leaf nodes: hold an actual character (like 'a' or 'b') and its frequency.
 *   - Internal nodes: created by merging two smaller nodes together. They hold
 *     the combined frequency but no real character (symbol is set to '\0').
 * 
 * When traversing the tree to find a code, going left adds a '0' bit
 * and going right adds a '1' bit.
 */
public class HuffmanNode {

    /** The character this node represents. Only meaningful for leaf nodes. */
    final char symbol;

    /** Frequency of this character, or the sum of both children if this is an internal node. */
    final int frequency;

    /** Left child — going here adds a '0' to the code. */
    HuffmanNode left;

    /** Right child — going here adds a '1' to the code. */
    HuffmanNode right;

    /** Use this constructor to create a leaf node that holds a real character. */
    HuffmanNode(char symbol, int frequency) {
        this.symbol    = symbol;
        this.frequency = frequency;
        this.left      = null;
        this.right     = null;
    }

    /** Use this constructor to create an internal node after merging two children. */
    HuffmanNode(int frequency, HuffmanNode left, HuffmanNode right) {
        this.symbol    = '\0'; // internal nodes don't represent any character
        this.frequency = frequency;
        this.left      = left;
        this.right     = right;
    }

    /** Returns true if this is a leaf node (no children = it holds a real character). */
    boolean isLeafNodeWithNoChildren() {
        return left == null && right == null;
    }
}

package huffman;

/**
 * This is a custom min-heap that stores HuffmanNode objects.
 * I needed it for Huffman tree construction because the algorithm always
 * needs to grab the two lowest-frequency nodes at each step.
 * 
 * The heap stores everything in a plain array. The node with the lowest
 * frequency always sits at index 0 (the root), so extracting it is O(log n).
 * 
 * Array index relationships:
 *   Parent of index i  →  (i - 1) / 2
 *   Left child of i    →  2*i + 1
 *   Right child of i   →  2*i + 2
 * 
 * When two nodes have the same frequency, I use two extra tiebreaker rules
 * to make the tree shape consistent and predictable:
 *   - Leaf nodes come before internal (merged) nodes
 *   - Among leaves, alphabetical order (lower ASCII value first)
 */
public class MinHeap {

    private HuffmanNode[] data;
    private int            size;
    private static final int INITIAL_CAPACITY = 16;

    MinHeap() {
        data = new HuffmanNode[INITIAL_CAPACITY];
        size = 0;
    }

    /** Returns how many nodes are currently in the heap. */
    int getCurrentHeapSize()       { return size; }

    /** Returns true if the heap has nothing in it. */
    boolean isHeapCurrentlyEmpty() { return size == 0; }

    /**
     * Adds a HuffmanNode to the heap and bubbles it up to the right position.
     * Time: O(log n)
     */
    void insertNodeIntoMinHeap(HuffmanNode node) {
        if (size == data.length) doubleHeapArrayCapacity();
        data[size] = node;
        siftNodeUpToRestoreHeapProperty(size);
        size++;
    }

    /**
     * Removes and returns the node with the lowest frequency (the root).
     * The last element in the array is moved to the root and then sinks down.
     * Returns null if the heap is empty.
     * Time: O(log n)
     */
    HuffmanNode removeAndReturnMinimumFrequencyNode() {
        if (size == 0) return null;
        HuffmanNode top = data[0];
        size--;
        data[0] = data[size];
        data[size] = null; // let GC reclaim memory
        if (size > 0) siftNodeDownToRestoreHeapProperty(0);
        return top;
    }

    /**
     * Returns the lowest-frequency node without removing it.
     * Returns null if the heap is empty.
     * Time: O(1)
     */
    HuffmanNode peekAtMinimumFrequencyNodeWithoutRemoving() {
        return (size == 0) ? null : data[0];
    }

    // -- Private heap mechanics --

    /**
     * Bubbles the node at 'index' upward until the heap rule is satisfied.
     * Called after inserting a new node at the end of the array.
     */
    private void siftNodeUpToRestoreHeapProperty(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;
            if (compareNodesByFrequencyThenTypeAndSymbol(data[index], data[parent]) < 0) {
                swapTwoNodesInHeapArray(index, parent);
                index = parent;
            } else {
                break;
            }
        }
    }

    /**
     * Sinks the node at 'index' downward until the heap rule is satisfied.
     * Called after removing the root and placing the last node there.
     */
    private void siftNodeDownToRestoreHeapProperty(int index) {
        while (true) {
            int left     = 2 * index + 1;
            int right    = 2 * index + 2;
            int smallest = index;

            if (left  < size && compareNodesByFrequencyThenTypeAndSymbol(data[left],  data[smallest]) < 0) smallest = left;
            if (right < size && compareNodesByFrequencyThenTypeAndSymbol(data[right], data[smallest]) < 0) smallest = right;

            if (smallest == index) break;
            swapTwoNodesInHeapArray(index, smallest);
            index = smallest;
        }
    }

    /**
     * Compares two nodes. Returns a negative value if node 'a' should
     * be extracted before node 'b' (i.e., 'a' has lower priority in the min-heap).
     * 
     * Rule 1: lower frequency comes first (min-heap).
     * Rule 2: if equal frequency, leaf nodes come before internal nodes.
     * Rule 3: if both are leaves with equal frequency, alphabetical order.
     */
    private static int compareNodesByFrequencyThenTypeAndSymbol(HuffmanNode a, HuffmanNode b) {
        if (a.frequency != b.frequency) {
            return a.frequency - b.frequency;
        }
        // Tiebreaker 1: leaf beats internal node
        boolean aLeaf = a.isLeafNodeWithNoChildren();
        boolean bLeaf = b.isLeafNodeWithNoChildren();
        if (aLeaf && !bLeaf) return -1;
        if (!aLeaf && bLeaf) return  1;
        // Tiebreaker 2: alphabetical among leaves
        if (aLeaf && bLeaf) return a.symbol - b.symbol;
        return 0;
    }

    /** Swaps two nodes at positions i and j in the array. */
    private void swapTwoNodesInHeapArray(int i, int j) {
        HuffmanNode temp = data[i];
        data[i] = data[j];
        data[j] = temp;
    }

    /** Doubles the array size when the heap is full. */
    private void doubleHeapArrayCapacity() {
        HuffmanNode[] larger = new HuffmanNode[data.length * 2];
        for (int i = 0; i < data.length; i++) larger[i] = data[i];
        data = larger;
    }
}

package huffman;

/**
 * This class builds the Huffman tree and handles both encoding and decoding.
 * 
 * How the tree is built (Huffman's greedy algorithm):
 *   1. Start by putting each character into the min-heap as a leaf node.
 *   2. Repeatedly pull out the two lowest-frequency nodes, merge them into
 *      a new internal node whose frequency is the sum of both, and put it back.
 *   3. Keep going until only one node is left — that is the root.
 * 
 * After building the tree, I do a pre-order traversal to assign codes.
 * Going left adds '0' to the code string, going right adds '1'.
 * 
 * There's a special case: if the input only has one unique character, the tree
 * has no edges, so I wrap it in a synthetic root to give it the code "0".
 * 
 * Time complexity:
 *   Tree construction  : O(n log n) — n heap operations, each O(log n)
 *   Encoding a message : O(m)       — one lookup per character
 *   Decoding bits      : O(b)       — one tree step per bit
 */
public class HuffmanTree {

    private HuffmanNode root;

    // Two parallel arrays to map characters to their Huffman bit codes.
    // I used parallel arrays instead of a map since we can't use java.util.
    private char[]   codeChars;
    private String[] codes;
    private int      codeCount;

    /**
     * Builds the Huffman tree from the given character and frequency arrays.
     * The 'count' tells us how many entries in the arrays are actually valid.
     */
    HuffmanTree(char[] characters, int[] frequencies, int count) {
        codeChars  = new char[count + 1];   // +1 covers the single-character edge case
        codes      = new String[count + 1];
        codeCount  = 0;

        if (count == 0) {
            root = null;
            return;
        }

        // Load every character as a leaf node into the min-heap.
        MinHeap heap = new MinHeap();
        for (int i = 0; i < count; i++) {
            heap.insertNodeIntoMinHeap(new HuffmanNode(characters[i], frequencies[i]));
        }

        // Edge case: only one unique character in the input.
        // Wrap it in an internal root so the tree has at least one edge.
        if (heap.getCurrentHeapSize() == 1) {
            HuffmanNode only = heap.removeAndReturnMinimumFrequencyNode();
            root = new HuffmanNode(only.frequency, only, null);
        } else {
            // Main greedy merge loop: always merge the two smallest nodes.
            while (heap.getCurrentHeapSize() > 1) {
                HuffmanNode left  = heap.removeAndReturnMinimumFrequencyNode();
                HuffmanNode right = heap.removeAndReturnMinimumFrequencyNode();
                HuffmanNode merged = new HuffmanNode(
                    left.frequency + right.frequency, left, right);
                heap.insertNodeIntoMinHeap(merged);
            }
            root = heap.removeAndReturnMinimumFrequencyNode();
        }

        // Walk the tree to assign a bit string to each character.
        generateHuffmanCodesViaPreOrderTraversal(root, "");
    }

    /**
     * Recursively walks the tree in pre-order.
     * When we reach a leaf, the accumulated prefix string is that character's code.
     * Left child always gets '0' added, right child gets '1'.
     */
    private void generateHuffmanCodesViaPreOrderTraversal(HuffmanNode node, String prefix) {
        if (node == null) return;
        if (node.isLeafNodeWithNoChildren()) {
            // Assign "0" if the prefix is empty (only happens in the single-char case).
            codeChars[codeCount] = node.symbol;
            codes[codeCount]     = prefix.isEmpty() ? "0" : prefix;
            codeCount++;
            return;
        }
        generateHuffmanCodesViaPreOrderTraversal(node.left,  prefix + "0");
        generateHuffmanCodesViaPreOrderTraversal(node.right, prefix + "1");
    }

    /** Returns true if the tree was built successfully (root is not null). */
    boolean isHuffmanTreeSuccessfullyBuilt() { return root != null; }

    /**
     * Converts a text message into a binary bit string using the Huffman codes.
     * Returns null if any character in the message is not in the tree.
     */
    String encodeStringToBinaryBitSequence(String message) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < message.length(); i++) {
            String code = lookUpHuffmanCodeForCharacter(message.charAt(i));
            if (code == null) return null; // character not in this tree
            sb.append(code);
        }
        return sb.toString();
    }

    /**
     * Converts a binary bit string back to the original text by walking the tree.
     * '0' means go left, '1' means go right. When we land on a leaf, that is
     * the next character. Then we reset to the root and continue.
     * Returns null if the bit string does not match any valid sequence in the tree.
     */
    String decodeBinaryBitSequenceToString(String encodedBits) {
        if (root == null) return "";

        // Handle the special single-character tree (root has only a left child).
        if (root.left != null && root.left.isLeafNodeWithNoChildren() && root.right == null) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < encodedBits.length(); i++) {
                if (encodedBits.charAt(i) == '0') {
                    sb.append(root.left.symbol);
                } else {
                    return null; // only '0' is valid for a single-char tree
                }
            }
            return sb.toString();
        }

        // Normal decoding: step through the tree one bit at a time.
        StringBuilder result  = new StringBuilder();
        HuffmanNode   current = root;
        for (int i = 0; i < encodedBits.length(); i++) {
            char bit = encodedBits.charAt(i);
            if      (bit == '0') current = current.left;
            else if (bit == '1') current = current.right;
            else return null; // invalid character in bit string

            if (current == null) return null; // path doesn't exist in the tree

            if (current.isLeafNodeWithNoChildren()) {
                result.append(current.symbol);
                current = root; // reset to root for the next character
            }
        }
        // If we stopped in the middle of a code, the bit string is incomplete.
        if (current != root) return null;
        return result.toString();
    }

    /** Prints the complete character-to-code table to the console. */
    void printHuffmanCodeLookupTable() {
        System.out.println("  Huffman Codes (Pre-order Traversal, left='0', right='1'):");
        printHorizontalDividerLine();
        System.out.printf("  %-14s %s%n", "Character", "Code");
        printHorizontalDividerLine();
        for (int i = 0; i < codeCount; i++) {
            String label = (codeChars[i] == ' ') ? "SPACE" : String.valueOf(codeChars[i]);
            System.out.printf("  %-14s %s%n", "'" + label + "'", codes[i]);
        }
        printHorizontalDividerLine();
    }

    /**
     * Searches through the code table to find the bit string for a given character.
     * Returns null if the character is not in the table.
     * Using a linear search here is fine since the number of unique characters is small.
     */
    private String lookUpHuffmanCodeForCharacter(char c) {
        for (int i = 0; i < codeCount; i++) {
            if (codeChars[i] == c) return codes[i];
        }
        return null;
    }

    private static void printHorizontalDividerLine() {
        System.out.println("  ─────────────────────────────────────────────────────────────────────");
    }
}

package logistics;

/**
 * Represents one directed shipping lane (edge) in the network.
 * 
 * Each lane has a baseWeight that is set when it is created and never changes.
 * The currentWeight starts equal to the baseWeight but gets adjusted after
 * every journey leg to simulate network congestion and recovery.
 * 
 * The three rules I implemented:
 *   Congestion : currentWeight *= 1.10  (used lanes get 10% heavier)
 *   Recovery   : currentWeight *= 0.95  (unused lanes recover by 5%, but not below base)
 *   Blocking   : if currentWeight exceeds 5x the base weight, the lane is
 *                permanently blocked and removed from future path calculations
 */
public class ShippingLane {

    static final double MAX_LATENCY_MULTIPLIER = 5.0; // block threshold
    static final double CONGESTION_RATE        = 0.10; // +10% when used
    static final double RECOVERY_RATE          = 0.05; // -5% when not used

    /** Name of the source port. */
    final String fromPort;

    /** Name of the destination port. */
    final String toPort;

    /** The original weight — never changes, acts as the recovery floor. */
    final double baseWeight;

    /** The current working weight used in Dijkstra. Changes after each journey leg. */
    double currentWeight;

    /** Once this is true, the lane is completely ignored by Dijkstra. */
    boolean isBlocked;

    ShippingLane(String from, String to, double baseWeight) {
        this.fromPort      = from;
        this.toPort        = to;
        this.baseWeight    = baseWeight;
        this.currentWeight = baseWeight; // start at the base
        this.isBlocked     = false;
    }

    /**
     * Applies the +10% congestion increase to this lane's current weight.
     * After increasing, checks if the blocking threshold has been crossed.
     * Does nothing if the lane is already blocked.
     */
    void applyCongestionWeightIncrease() {
        if (isBlocked) return;
        currentWeight *= (1.0 + CONGESTION_RATE);
        checkThresholdAndPermanentlyBlockLaneIfExceeded();
    }

    /**
     * Applies the -5% recovery decrease to this lane's current weight.
     * The weight is clamped so it never drops below the original baseWeight.
     * Does nothing if the lane is already blocked.
     */
    void applyRecoveryWeightDecrease() {
        if (isBlocked) return;
        currentWeight *= (1.0 - RECOVERY_RATE);
        if (currentWeight < baseWeight) currentWeight = baseWeight; // floor at base
    }

    /**
     * Checks if the current weight has exceeded 5 times the base weight.
     * If so, the lane is permanently blocked — no further updates will affect it.
     */
    private void checkThresholdAndPermanentlyBlockLaneIfExceeded() {
        if (currentWeight > baseWeight * MAX_LATENCY_MULTIPLIER) {
            isBlocked = true;
        }
    }
}

package logistics;

/**
 * A simple container to return the result of one Dijkstra run.
 * 
 * It holds three things the caller needs after computeShortestPathUsingDijkstra():
 *   - path      : the sequence of port indices from source to destination
 *   - totalCost : the total weight of the chosen path
 *   - usedLanes : a boolean array where usedLanes[i] is true if lane i was
 *                 part of the path (used to decide which lanes get congestion penalty)
 */
class DijkstraResult {

    final int[]     path;
    final double    totalCost;
    final boolean[] usedLanes;

    DijkstraResult(int[] path, double totalCost, boolean[] usedLanes) {
        this.path      = path;
        this.totalCost = totalCost;
        this.usedLanes = usedLanes;
    }

package logistics;

/**
 * Custom min-heap used specifically for Dijkstra's algorithm.
 * Instead of creating wrapper objects, I used two parallel arrays —
 * one for distances and one for port indices — to keep memory usage low.
 * 
 * The smallest distance always sits at index 0, so extracting it is O(log n).
 * 
 * Lazy deletion: if a port is already visited when it comes out of the heap,
 * DijkstraEngine just skips it. This means we don't need a decrease-key
 * operation, which makes the code much simpler.
 */
class DijkstraMinHeap {

    private double[] distances;
    private int[]    portIndices;
    private int      size;
    private static final int INITIAL_CAPACITY = 32;

    DijkstraMinHeap() {
        distances   = new double[INITIAL_CAPACITY];
        portIndices = new int[INITIAL_CAPACITY];
        size        = 0;
    }

    /** Returns true if the heap has no entries. */
    boolean isDijkstraHeapCurrentlyEmpty() { return size == 0; }

    /**
     * Adds a (distance, portIndex) pair to the heap and bubbles it up.
     * Time: O(log n)
     */
    void insertDistanceEntryIntoHeap(double dist, int portIdx) {
        if (size == distances.length) doubleHeapArraysCapacity();
        distances[size]   = dist;
        portIndices[size] = portIdx;
        siftEntryUpToRestoreMinHeapOrder(size);
        size++;
    }

    /**
     * Removes and returns the entry with the smallest distance.
     * Returns null if the heap is empty.
     * Time: O(log n)
     */
    DijkstraEntry removeAndReturnSmallestDistanceEntry() {
        if (size == 0) return null;
        double d = distances[0];
        int    i = portIndices[0];
        size--;
        distances[0]   = distances[size];
        portIndices[0] = portIndices[size];
        if (size > 0) siftEntryDownToRestoreMinHeapOrder(0);
        return new DijkstraEntry(d, i);
    }

    /**
     * Bubbles the entry at index k upward until the min-heap rule is satisfied.
     * Used after inserting a new entry at the end.
     */
    private void siftEntryUpToRestoreMinHeapOrder(int k) {
        while (k > 0) {
            int parent = (k - 1) / 2;
            if (distances[k] < distances[parent]) {
                swapTwoHeapEntriesAtGivenIndices(k, parent);
                k = parent;
            } else break;
        }
    }

    /**
     * Sinks the entry at index k downward until the min-heap rule is satisfied.
     * Used after removing the root entry.
     */
    private void siftEntryDownToRestoreMinHeapOrder(int k) {
        while (true) {
            int left = 2 * k + 1, right = 2 * k + 2, smallest = k;
            if (left  < size && distances[left]  < distances[smallest]) smallest = left;
            if (right < size && distances[right] < distances[smallest]) smallest = right;
            if (smallest == k) break;
            swapTwoHeapEntriesAtGivenIndices(k, smallest);
            k = smallest;
        }
    }

    /** Swaps the distance and portIndex values at positions i and j. */
    private void swapTwoHeapEntriesAtGivenIndices(int i, int j) {
        double td = distances[i];   distances[i]   = distances[j];   distances[j]   = td;
        int    ti = portIndices[i]; portIndices[i] = portIndices[j]; portIndices[j] = ti;
    }

    /** Doubles the capacity of both arrays when the heap fills up. */
    private void doubleHeapArraysCapacity() {
        double[] nd = new double[distances.length * 2];
        int[]    ni = new int[portIndices.length * 2];
        for (int i = 0; i < distances.length; i++) { nd[i] = distances[i]; ni[i] = portIndices[i]; }
        distances = nd; portIndices = ni;
    }
}

/**
 * Simple holder class that bundles a distance and a port index together.
 * Used as the return type when extracting from DijkstraMinHeap.
 */
class DijkstraEntry {
    final double distance;
    final int    portIdx;
    DijkstraEntry(double distance, int portIdx) {
        this.distance = distance;
        this.portIdx  = portIdx;
    }
}

package logistics;

/**
 * This class runs Dijkstra's shortest path algorithm on the shipping network.
 * 
 * I used a custom min-heap (DijkstraMinHeap) to always pick the port with
 * the smallest known distance next. This gives O(log V) extraction instead
 * of the O(V) scan you would need with a plain array.
 * 
 * I also used "lazy deletion" — when a port is settled (visited), I just
 * skip it if it comes up again in the heap instead of removing it.
 * This avoids needing a decrease-key operation.
 * 
 * The algorithm stops early as soon as the destination port is settled,
 * so it doesn't explore the whole graph unnecessarily.
 * 
 * Overall time: O((V + E) log V)
 *   V extractions from the heap: each O(log V)
 *   E edge relaxations total: each O(log V) for the heap insert
 * 
 * One important requirement: all edge weights must be non-negative.
 * This is guaranteed here because base weights are always positive and
 * the congestion/recovery rules never bring a weight below the base.
 */
public class DijkstraEngine {

    private final ShippingGraph graph;

    DijkstraEngine(ShippingGraph graph) {
        this.graph = graph;
    }

    /**
     * Finds the shortest path from sourceIdx to destIdx in the current graph state.
     * 
     * Returns a DijkstraResult containing:
     *   - the sequence of port indices from source to destination
     *   - the total cost of that path
     *   - which lanes were used (so the caller can apply weight updates)
     * 
     * Returns null if the destination is unreachable (all paths are blocked).
     */
    DijkstraResult computeShortestPathUsingDijkstra(int sourceIdx, int destIdx) {

        int n = graph.getTotalRegisteredPortCount();
        if (n == 0) return null;

        double[]      dist    = new double[n];
        int[]         prev    = new int[n];
        boolean[]     visited = new boolean[n];
        ShippingLane[] laneTo = new ShippingLane[n]; // tracks which lane was used to reach each port

        // Initialise all distances to infinity except the source.
        for (int i = 0; i < n; i++) {
            dist[i] = Double.MAX_VALUE;
            prev[i] = -1;
        }
        dist[sourceIdx] = 0.0;

        DijkstraMinHeap pq = new DijkstraMinHeap();
        pq.insertDistanceEntryIntoHeap(0.0, sourceIdx);

        // Main loop: keep pulling out the closest unvisited port.
        while (!pq.isDijkstraHeapCurrentlyEmpty()) {
            DijkstraEntry top = pq.removeAndReturnSmallestDistanceEntry();
            int u = top.portIdx;

            if (visited[u]) continue; // lazy deletion: skip stale entries
            visited[u] = true;

            if (u == destIdx) break; // we've found the shortest path to the destination

            // Try to relax all edges leaving port u.
            EdgeNode node = graph.getAdjacencyListHeadForPort(u);
            while (node != null) {
                ShippingLane lane = node.lane;
                if (!lane.isBlocked) {
                    int    v       = node.toPortIdx;
                    double newDist = dist[u] + lane.currentWeight;
                    if (newDist < dist[v]) {
                        dist[v]   = newDist;
                        prev[v]   = u;
                        laneTo[v] = lane;
                        pq.insertDistanceEntryIntoHeap(newDist, v);
                    }
                }
                node = node.next;
            }
        }

        // If the destination still has infinity distance, it was unreachable.
        if (dist[destIdx] == Double.MAX_VALUE) return null;

        // Reconstruct the path by backtracking through the prev[] array.
        int pathLen = 0;
        for (int cur = destIdx; cur != -1; cur = prev[cur]) pathLen++;

        int[] path = new int[pathLen];
        int   pos  = pathLen - 1;
        for (int cur = destIdx; cur != -1; cur = prev[cur]) {
            path[pos--] = cur;
        }

        // Build an array marking which lanes were actually used in this path.
        int       totalLanes = graph.getTotalShippingLaneCount();
        boolean[] usedLanes  = new boolean[totalLanes];

        for (int i = 1; i < path.length; i++) {
            ShippingLane lane = laneTo[path[i]];
            if (lane != null) {
                int idx = graph.findIndexOfShippingLaneInArray(lane);
                if (idx != -1) usedLanes[idx] = true;
            }
        }

        return new DijkstraResult(path, dist[destIdx], usedLanes);
    }
}
PartB_Scenario02_Logistics/ShippingGraph.java 
package logistics;

/**
 * This is the directed weighted graph that models the shipping network.
 * 
 * Each port is a vertex and each shipping lane is a directed edge.
 * I chose an adjacency list over an adjacency matrix because shipping
 * networks are sparse — each port only connects to a few others, not all of them.
 * This saves memory: O(V + E) instead of O(V²).
 * 
 * The edges are stored as singly-linked lists (using EdgeNode) with one list
 * per port. New edges are prepended at the head in O(1).
 * 
 * I also keep a flat allLanes[] array alongside the adjacency list so that
 * the adaptive weight update can iterate through all lanes efficiently
 * without traversing the whole graph structure.
 * 
 * Limits: up to 100 ports and 500 lanes (expandable by changing the constants).
 */
public class ShippingGraph {

    private static final int MAX_PORTS = 100;
    private static final int MAX_LANES = 500;

    private String[]     portNames;  // portNames[i] = the name of port i
    private int          portCount;

    /**
     * The adjacency list. adjacency[i] is the head of the linked list of
     * outgoing edges from port i.
     */
    private EdgeNode[]   adjacency;

    /** Flat array of all lanes for global weight updates and audits. */
    private ShippingLane[] allLanes;
    private int            laneCount;

    ShippingGraph() {
        portNames  = new String[MAX_PORTS];
        adjacency  = new EdgeNode[MAX_PORTS];
        portCount  = 0;
        allLanes   = new ShippingLane[MAX_LANES];
        laneCount  = 0;
    }

    // -- Port operations --

    /**
     * Adds a new port to the graph as a vertex.
     * If the port already exists, just returns its existing index.
     * Returns -1 if the port limit has been reached.
     */
    int registerNewPortIntoGraph(String name) {
        int existing = findPortIndexByName(name);
        if (existing != -1) return existing;
        if (portCount >= MAX_PORTS) return -1;
        portNames[portCount] = name;
        adjacency[portCount] = null;
        return portCount++;
    }

    /**
     * Searches for a port by name and returns its index.
     * The search is case-insensitive.
     * Returns -1 if no port with that name is found.
     */
    int findPortIndexByName(String name) {
        for (int i = 0; i < portCount; i++) {
            if (portNames[i].equalsIgnoreCase(name)) return i;
        }
        return -1;
    }

    /** Returns the name of the port at the given index, or null if out of range. */
    String getPortNameByIndex(int idx) {
        return (idx >= 0 && idx < portCount) ? portNames[idx] : null;
    }

    /** Returns how many ports are currently registered. */
    int getTotalRegisteredPortCount() { return portCount; }

    // -- Lane (edge) operations --

    /**
     * Adds a directed shipping lane from one port to another with the given base weight.
     * If either port doesn't exist yet, it gets registered automatically.
     * Returns false if the input is invalid or the lane limit is reached.
     */
    boolean addDirectedShippingLaneBetweenPorts(String from, String to, double weight) {
        if (weight <= 0)                        return false;
        if (from.equalsIgnoreCase(to))          return false;
        if (laneCount >= MAX_LANES)             return false;

        int fromIdx = registerNewPortIntoGraph(from);
        int toIdx   = registerNewPortIntoGraph(to);
        if (fromIdx == -1 || toIdx == -1)       return false;

        ShippingLane lane    = new ShippingLane(from, to, weight);
        EdgeNode     newNode = new EdgeNode(lane, toIdx);

        // Prepend to the adjacency list — O(1) head insertion.
        newNode.next       = adjacency[fromIdx];
        adjacency[fromIdx] = newNode;

        allLanes[laneCount++] = lane;
        return true;
    }

    /** Returns the head of the adjacency list for the given port index. */
    EdgeNode getAdjacencyListHeadForPort(int portIdx) {
        return (portIdx >= 0 && portIdx < portCount) ? adjacency[portIdx] : null;
    }

    // -- Adaptive weight update (called after each Dijkstra leg) --

    /**
     * Goes through every lane in the network and applies the weight update rules:
     *   - Lanes that were used in the last journey leg get +10% congestion.
     *   - Lanes that were NOT used get -5% recovery (but never below baseWeight).
     * 
     * The usedFlags array is parallel to allLanes[]: usedFlags[i] is true if
     * lane i was part of the computed path.
     */
    void applyCongestionAndRecoveryWeightUpdates(boolean[] usedFlags) {
        int limit = Math.min(laneCount, usedFlags.length);
        for (int i = 0; i < limit; i++) {
            if (allLanes[i].isBlocked) continue;
            if (usedFlags[i]) {
                allLanes[i].applyCongestionWeightIncrease();
            } else {
                allLanes[i].applyRecoveryWeightDecrease();
            }
        }
    }

    // -- Accessors for DijkstraEngine --

    /** Returns the total number of shipping lanes currently in the graph. */
    int getTotalShippingLaneCount()      { return laneCount; }

    /** Returns a copy of the allLanes array (does not expose the internal array). */
    ShippingLane[] getAllShippingLanesCopy() {
        ShippingLane[] copy = new ShippingLane[laneCount];
        for (int i = 0; i < laneCount; i++) copy[i] = allLanes[i];
        return copy;
    }

    /**
     * Finds the index of a specific lane object in allLanes[].
     * Uses reference equality (==) to match the exact object.
     * Returns -1 if not found.
     */
    int findIndexOfShippingLaneInArray(ShippingLane lane) {
        for (int i = 0; i < laneCount; i++) {
            if (allLanes[i] == lane) return i;
        }
        return -1;
    }

    // -- Display --

    /** Prints a live table of all shipping lanes showing their current weights and status. */
    void displayLiveNetworkAuditOfAllLanes() {
        System.out.println();
        System.out.println("  ╔══════════════════ LIVE NETWORK AUDIT ══════════════════╗");
        System.out.printf("  ║  %-14s %-14s %8s %12s %10s ║%n",
            "From", "To", "Base", "Current", "Status");
        System.out.println("  ╠════════════════════════════════════════════════════════╣");
        int blockedCount = 0;
        for (int i = 0; i < laneCount; i++) {
            ShippingLane s = allLanes[i];
            String status  = s.isBlocked ? "BLOCKED" : "active";
            String current = s.isBlocked ? "N/A" : String.format("%.2f", s.currentWeight);
            System.out.printf("  ║  %-14s %-14s %8.2f %12s %10s ║%n",
                s.fromPort, s.toPort, s.baseWeight, current, status);
            if (s.isBlocked) blockedCount++;
        }
        System.out.println("  ╚════════════════════════════════════════════════════════╝");
        System.out.printf("  Summary: %d total lanes | %d blocked | %d active%n%n",
            laneCount, blockedCount, laneCount - blockedCount);
    }
}
